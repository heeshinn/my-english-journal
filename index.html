const SHEET_ID = '1STFpojqhYNiVOhXHY7TRU7nixq7C474YnMalJBsXGhs'; const SHEET_NAME = 'PHRASES'; function doGet(e) { const action = e.parameter.action; if (action === 'loadPhrases') return loadPhrases(); return ContentService.createTextOutput(JSON.stringify({error:'Unknown action'})).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const data = JSON.parse(e.postData.contents); if (data.action === 'savePhrases') return savePhrases(data.phrases); if (data.action === 'updateStats') return updateStats(data.phraseId, data.correct); return ContentService.createTextOutput(JSON.stringify({error:'Unknown action'})).setMimeType(ContentService.MimeType.JSON); } function savePhrases(phrases) { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['id','date','type','topic','original','english','expression','korean','difficulty','review_count','wrong_count','source']); } const existing = sheet.getDataRange().getValues(); const ids = existing.slice(1).map(r => r[0]); phrases.forEach(p => { const row = [p.id, p.date||'', p.type||'', p.topic||'', p.original||'', p.english||'', p.expression||'', p.korean||'', p.difficulty||'', 0, 0, p.source||'']; const idx = ids.indexOf(p.id); if (idx >= 0) sheet.getRange(idx+2, 1, 1, row.length).setValues([row]); else sheet.appendRow(row); }); return ContentService.createTextOutput(JSON.stringify({ok:true, count:phrases.length})).setMimeType(ContentService.MimeType.JSON); } function loadPhrases() { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return ContentService.createTextOutput(JSON.stringify({phrases:[]})).setMimeType(ContentService.MimeType.JSON); const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const phrases = rows.slice(1).map(r => { const obj = {}; headers.forEach((h,i) => obj[h] = r[i]); return obj; }); return ContentService.createTextOutput(JSON.stringify({phrases})).setMimeType(ContentService.MimeType.JSON); } function updateStats(phraseId, correct) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return ContentService.createTextOutput(JSON.stringify({ok:false})).setMimeType(ContentService.MimeType.JSON); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === phraseId) { sheet.getRange(i+1, 10).setValue((rows[i][9]||0)+1); if (!correct) sheet.getRange(i+1, 11).setValue((rows[i][10]||0)+1); break; } } return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON); }
